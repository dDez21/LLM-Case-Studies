def ceil2(value: float) -> float:
    """Arrotondamento per eccesso ai centesimi."""
    return math.ceil(value * 100) / 100


def lookup(name: str, qty_g: float) -> dict | None:
    """Cerca l'ingrediente nel DB e scala i valori alla quantità indicata."""
    key = name.strip().lower()
    # Ricerca esatta
    if key in NUTRIENT_DB:
        data = NUTRIENT_DB[key]
        factor = qty_g / 100.0
        return {
            "name":     name.strip(),
            "quantity": qty_g,
            "calories": ceil2(data[0] * factor),
            "proteins": ceil2(data[1] * factor),
            "lipids":   ceil2(data[2] * factor),
            "sodium":   ceil2(data[3] * factor),
            "fibers":   ceil2(data[4] * factor),
        }
    # Ricerca parziale (il nome del DB è contenuto nell'input o viceversa)
    for db_key, data in NUTRIENT_DB.items():
        if db_key in key or key in db_key:
            factor = qty_g / 100.0
            return {
                "name":     name.strip(),
                "quantity": qty_g,
                "calories": ceil2(data[0] * factor),
                "proteins": ceil2(data[1] * factor),
                "lipids":   ceil2(data[2] * factor),
                "sodium":   ceil2(data[3] * factor),
                "fibers":   ceil2(data[4] * factor),
            }
    return None

    def _add_row(self):
        if len(self.ingredient_rows) >= 15:
            self._set_status("⚠  Massimo 15 ingredienti raggiunto.", COLORS["warning"])
            return

        idx = len(self.ingredient_rows)
        grid_row = idx + 1   # row 0 is the header

        row_frame = ctk.CTkFrame(self.scroll, fg_color="transparent")
        row_frame.grid(row=grid_row, column=0, columnspan=4,
                       sticky="ew", padx=2, pady=2)
        row_frame.grid_columnconfigure(0, weight=0)
        row_frame.grid_columnconfigure(1, weight=3)
        row_frame.grid_columnconfigure(2, weight=2)
        row_frame.grid_columnconfigure(3, weight=0)

        num_lbl = ctk.CTkLabel(
            row_frame, text=f"{idx + 1:02d}",
            font=ctk.CTkFont("Segoe UI", 12),
            text_color=COLORS["text_secondary"], width=30
        )
        num_lbl.grid(row=0, column=0, padx=(4, 2), sticky="e")

        ing_entry = ctk.CTkEntry(
            row_frame,
            placeholder_text="es. Riso, Pollo…",
            font=ctk.CTkFont("Segoe UI", 13),
            fg_color="#1e2133", border_color=COLORS["border"],
            text_color=COLORS["text_primary"],
            placeholder_text_color=COLORS["text_secondary"],
            border_width=1, corner_radius=8, height=34
        )
        ing_entry.grid(row=0, column=1, padx=4, sticky="ew")

        qty_entry = ctk.CTkEntry(
            row_frame,
            placeholder_text="g",
            font=ctk.CTkFont("Segoe UI", 13),
            fg_color="#1e2133", border_color=COLORS["border"],
            text_color=COLORS["text_primary"],
            placeholder_text_color=COLORS["text_secondary"],
            border_width=1, corner_radius=8, height=34, width=85
        )
        qty_entry.grid(row=0, column=2, padx=4, sticky="w")

        del_btn = ctk.CTkButton(
            row_frame, text="✕", width=30, height=30,
            font=ctk.CTkFont("Segoe UI", 12, "bold"),
            fg_color=COLORS["del_btn"],
            hover_color=COLORS["del_btn_hov"],

            command=lambda rf=row_frame: self._remove_row(rf)
        )
        del_btn.grid(row=0, column=3, padx=(4, 2))

        self.ingredient_rows.append((row_frame, ing_entry, qty_entry))
        self._refresh_numbers()

        if len(self.ingredient_rows) >= 15:
            self.add_btn.configure(state="disabled")

    def _remove_row(self, row_frame: ctk.CTkFrame):
        for i, (rf, _, _) in enumerate(self.ingredient_rows):
            if rf is row_frame:
                self.ingredient_rows.pop(i)
                rf.destroy()
                break
        self.add_btn.configure(state="normal")
        self._refresh_numbers()
        self._set_status("", COLORS["text_secondary"])

    def _refresh_numbers(self):
        """Aggiorna i numeri progressivi dopo aggiunta/rimozione."""
        self.count_lbl.configure(text=f"{len(self.ingredient_rows)} / 15")
        for i, (rf, _, _) in enumerate(self.ingredient_rows):
            # Il primo widget figlio nella colonna 0 è il num_lbl
            for child in rf.winfo_children():
                info = child.grid_info()
                if info.get("column") == 0:
                    child.configure(text=f"{i + 1:02d}")
                    break

                   sticky="ew", ipadx=6, ipady=6)

    def _draw_placeholder(self):

            self.table_scroll,
            text="Inserisci ingredienti e premi «⚡ Calcola»",
            font=ctk.CTkFont("Segoe UI", 13),
            text_color=COLORS["text_secondary"],
            justify="center"
        )
        self._placeholder.grid(row=1, column=0, columnspan=7, pady=70, sticky="ew")

    def _reset_table(self):
        """Svuota la tabella dei risultati e mostra il segnaposto."""
        for w in self.table_scroll.winfo_children():
            w.destroy()
        self._draw_table_header()
        self._draw_placeholder()

    # ── Calculation ───────────────────────────────────────────────────────────
    def _on_calculate(self):
        ingredients = []
        not_found = []

        for _, ing_e, qty_e in self.ingredient_rows:
            name = ing_e.get().strip()
            qty  = qty_e.get().strip()
            if not name and not qty:
                continue
            if not name:
                self._reset_table()
                self._set_status("⚠  Inserisci il nome di tutti gli ingredienti.", COLORS["warning"])
                return
            if not qty:
                self._reset_table()
                self._set_status(f"⚠  Quantità mancante per «{name}».", COLORS["warning"])
                return
            try:
                qty_val = float(qty.replace(",", "."))
            except ValueError:
                self._reset_table()
                self._set_status(f"⚠  Quantità non valida per «{name}».", COLORS["warning"])
                return
            if qty_val <= 0:
                self._reset_table()
                self._set_status(f"⚠  La quantità di «{name}» deve essere > 0.", COLORS["warning"])
                return

            result = lookup(name, qty_val)
            if result is None:
                not_found.append(name)
            else:
                ingredients.append(result)

        if not ingredients and not not_found:
            self._reset_table()
            self._set_status("⚠  Inserisci almeno un ingrediente.", COLORS["warning"])
            return

        if not_found:
            missing = ", ".join(not_found)
            if not ingredients:
                self._reset_table()
                self._set_status(f"⚠  Non trovati nel DB: {missing}", COLORS["warning"])
                return

        self._render_results(ingredients, not_found)

    # ── Rendering ─────────────────────────────────────────────────────────────
    def _render_results(self, rows: list[dict], not_found: list[str]):
        # Clear previous table content (keep header row=0)
        for w in self.table_scroll.winfo_children():
            w.destroy()
        self._draw_table_header()

        totals = {"quantity": 0.0, "calories": 0.0, "proteins": 0.0,
                  "lipids": 0.0, "sodium": 0.0, "fibers": 0.0}

        for i, row in enumerate(rows):
            bg = COLORS["bg_row_even"] if i % 2 == 0 else COLORS["bg_row_odd"]
            for key in totals:
                totals[key] += row[key]

            values = [
                row["name"],
                f"{row['quantity']:.0f}",
                f"{row['calories']:.2f}",
                f"{row['proteins']:.2f}",
                f"{row['lipids']:.2f}",
                f"{row['sodium']:.2f}",
                f"{row['fibers']:.2f}",
            ]
            for col, val in enumerate(values):
                ctk.CTkLabel(
                    self.table_scroll, text=val,
                    font=ctk.CTkFont("Segoe UI", 12),
                    fg_color=bg,
                    text_color=COLORS["text_primary"],
                    corner_radius=0,
                    anchor="w" if col == 0 else "center",
                    justify="left" if col == 0 else "center"
                ).grid(row=i + 1, column=col, padx=1, pady=1,
                       sticky="ew", ipadx=8, ipady=5)

        # Ceil totals as well
        for key in totals:
            totals[key] = ceil2(totals[key])

        # Totals row
        total_row = len(rows) + 1
        total_vals = [
            "TOTALE", f"{totals['quantity']:.0f}",
            f"{totals['calories']:.2f}",
            f"{totals['proteins']:.2f}",
            f"{totals['lipids']:.2f}",
            f"{totals['sodium']:.2f}",
            f"{totals['fibers']:.2f}",
        ]
        for col, val in enumerate(total_vals):
            ctk.CTkLabel(
                self.table_scroll, text=val,
                font=ctk.CTkFont("Segoe UI", 13, "bold"),
                fg_color=COLORS["bg_total"],
                text_color="white",
                corner_radius=4,
                anchor="w" if col == 0 else "center",
                justify="left" if col == 0 else "center"
            ).grid(row=total_row, column=col, padx=1, pady=(5, 2),
                   sticky="ew", ipadx=8, ipady=7)

        msg = f"✅  {len(rows)} ingredient{'e' if len(rows)==1 else 'i'} calcolat{'o' if len(rows)==1 else 'i'}."
        if not_found:
            msg += f"  ⚠ Non trovati: {', '.join(not_found)}"
        self._set_status(msg, COLORS["success"] if not not_found else COLORS["warning"])

    # ── Helpers ───────────────────────────────────────────────────────────────
    def _set_status(self, text: str, color: str):
        self.status_lbl.configure(text=text, text_color=color)
