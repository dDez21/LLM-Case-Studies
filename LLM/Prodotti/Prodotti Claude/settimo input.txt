BASE_URL = "https://api.nal.usda.gov/fdc/v1"

DATA_TYPE_PRIORITY = {"Foundation": 0, "SR Legacy": 1, "Survey (FNDDS)": 2, "Branded": 3}
MAX_SUGGESTIONS    = 10

# Nutrient IDs used by USDA (some have legacy + new IDs)
NUTRIENT_IDS: dict[str, set] = {
    "calories": {1008, 208},
    "proteins": {1003, 203},
    "lipids":   {1004, 204},
    "sodium":   {1093, 307},
    "fibers":   {1079, 291},
}

SIMPLE_WORDS  = {"raw", "fresh", "plain", "whole", "cooked", "boiled",
                 "steamed", "roasted", "grilled", "baked", "dried", "skinless"}
COMPLEX_WORDS = {"rotisserie", "breaded", "commercial", "frozen", "restaurant",
                 "fast", "takeout", "ready-to-eat", "seasoned", "prepared", "flavored"}


def _sort_key(food: dict) -> tuple:
    dt   = food.get("dataType", "Branded")
    prio = DATA_TYPE_PRIORITY.get(dt, 4)
    desc = food.get("description", "").lower()
    bad  =  sum(2 for w in COMPLEX_WORDS if w in desc)
    good = -sum(1 for w in SIMPLE_WORDS  if w in desc)
    return (prio, bad + good, len(desc))


def usda_search(query: str) -> list[dict]:
    """Search USDA FoodData Central; return sorted suggestions."""
    params = [
        ("query",    query),
        ("api_key",  API_KEY),
        ("pageSize", "30"),
        ("dataType", "Foundation"),
        ("dataType", "SR Legacy"),
        ("dataType", "Survey (FNDDS)"),
        ("dataType", "Branded"),
    ]
    url = f"{BASE_URL}/foods/search?" + urllib.parse.urlencode(params)
    with urllib.request.urlopen(url, timeout=10) as r:
        data = json.loads(r.read())
    return sorted(data.get("foods", []), key=_sort_key)[:MAX_SUGGESTIONS]


def usda_nutrients(fdc_id: int, qty_g: float) -> dict[str, float]:
    """Fetch nutrient detail for fdc_id and scale proportionally to qty_g."""
    url = f"{BASE_URL}/food/{fdc_id}?api_key={API_KEY}"
    with urllib.request.urlopen(url, timeout=10) as r:
        data = json.loads(r.read())

    raw: dict[str, float] = {k: 0.0 for k in NUTRIENT_IDS}
    for n in data.get("foodNutrients", []):
        # Handles both Foundation/SR Legacy and Branded response structures
        nobj = n.get("nutrient") or {}
        nid  = nobj.get("id") or n.get("nutrientId") or n.get("nutrientNumber")
        amt  = float(n.get("amount") or n.get("value") or 0)
        for key, ids in NUTRIENT_IDS.items():
            if nid in ids:
                raw[key] = amt
                break

    f = qty_g / 100.0
    return {k: math.ceil(v * f * 100) / 100 for k, v in raw.items()}


def fmt_qty(value: float) -> str:
    """Format a quantity without exponential notation and without trailing zeros."""
    return format(Decimal(str(value)).normalize(), "f")


# ─── Autocomplete dropdown ─────────────────────────────────────────────────────
class Dropdown(tk.Toplevel):
    """Floating suggestion list anchored below a CTkEntry widget."""

    ROW_H = 38

    def __init__(self, root: ctk.CTk, anchor: ctk.CTkEntry,
                 results: list[dict], on_select):
        super().__init__(root)
        self.overrideredirect(True)
        self.wm_attributes("-topmost", True)
        self.configure(bg=COLORS["border"])
        self._on_select = on_select
        self._build(anchor, results)

    def _build(self, anchor: ctk.CTkEntry, results: list[dict]) -> None:
        anchor.update_idletasks()
        w = max(anchor.winfo_width(), 300)
        h = min(len(results) * self.ROW_H + 2, 304)
        x = anchor.winfo_rootx()
        y = anchor.winfo_rooty() + anchor.winfo_height() + 2
        self.geometry(f"{w}x{h}+{x}+{y}")

        canvas = tk.Canvas(self, bg=COLORS["dd_bg"], highlightthickness=0, bd=0)
        if h >= 304:
            sb = tk.Scrollbar(self, orient="vertical", command=canvas.yview)
            canvas.configure(yscrollcommand=sb.set)
            sb.pack(side="right", fill="y")
        canvas.pack(fill="both", expand=True)

        inner = tk.Frame(canvas, bg=COLORS["dd_bg"])
        win   = canvas.create_window((0, 0), window=inner, anchor="nw")
        inner.bind("<Configure>", lambda e: canvas.configure(
            scrollregion=canvas.bbox("all")))
        canvas.bind("<Configure>", lambda e: canvas.itemconfigure(win, width=e.width))

        for i, food in enumerate(results):
            desc = food.get("description", "")
            bg   = COLORS["dd_even"] if i % 2 == 0 else COLORS["dd_bg"]
            btn  = tk.Button(
                inner, text=desc, anchor="w",
                bg=bg, fg=COLORS["text_primary"],
                activebackground=COLORS["dd_hover"], activeforeground="white",
                bd=0, padx=14, pady=8,
                font=("Segoe UI", 11), cursor="hand2",
                justify="left", wraplength=w - 28,
                command=lambda f=food: self._pick(f)
            )
            btn.pack(fill="x")
            btn.bind("<Enter>", lambda e, b=btn:       b.configure(bg=COLORS["dd_hover"]))
            btn.bind("<Leave>", lambda e, b=btn, c=bg: b.configure(bg=c))

    def _pick(self, food: dict) -> None:
        self._on_select(food)
        try:
            self.destroy()
        except tk.TclError:
            pass

    def _remove_row(self, row: dict) -> None:
        self._close_dropdown(row)
        if row in self.ingredient_rows:
            self.ingredient_rows.remove(row)
        row["frame"].destroy()
        self.add_btn.configure(state="normal")
        self._refresh_numbers()
        self._set_status("", COLORS["text_secondary"])

    def _refresh_numbers(self) -> None:
        self.count_lbl.configure(text=f"{len(self.ingredient_rows)} / 15")
        for i, row in enumerate(self.ingredient_rows):
            for child in row["frame"].winfo_children():
                if child.grid_info().get("column") == 0:
                    child.configure(text=f"{i + 1:02d}")
                    break

    # ── Search & dropdown ─────────────────────────────────────────────────────
    def _on_key(self, row: dict) -> None:
        text = row["ing_entry"].get().strip()

        # Invalidate confirmed selection if the user edits the text
        if row["fdc_id"] and text != row["confirmed_text"]:
            row["fdc_id"] = None
            row["confirmed_text"] = ""

        if row["debounce_job"]:
            self.after_cancel(row["debounce_job"])
            row["debounce_job"] = None

        if len(text) < 2:
            self._close_dropdown(row)
            return

        row["debounce_job"] = self.after(420, lambda q=text: self._do_search(row, q))

    def _do_search(self, row: dict, query: str) -> None:
        def _bg():
            try:
                results = usda_search(query)
                self.after(0, lambda: self._show_dropdown(row, results))
            except Exception:
                pass
        threading.Thread(target=_bg, daemon=True).start()

    def _show_dropdown(self, row: dict, results: list[dict]) -> None:
        self._close_dropdown(row)
        if not results:
            return

        ing = row["ing_entry"]

        def on_select(food: dict) -> None:
            row["fdc_id"]         = food["fdcId"]
            row["confirmed_text"] = food["description"]
            row["dropdown"]       = None
            ing.delete(0, "end")
            ing.insert(0, food["description"])

        try:
            row["dropdown"] = Dropdown(self, ing, results, on_select)
        except Exception:
            pass

    def _close_dropdown(self, row: dict) -> None:
        dd = row.get("dropdown")
        if dd:
            try:
                dd.destroy()
            except tk.TclError:
                pass
            row["dropdown"] = None

    def _close_all_dropdowns(self) -> None:
        for row in self.ingredient_rows:
            self._close_dropdown(row)

    def _draw_table_header(self) -> None:

        ctk.CTkLabel(
            self.table_scroll,
            text="Cerca e seleziona gli ingredienti dal database USDA,\npoi premi «⚡ Calcola»",
            font=ctk.CTkFont("Segoe UI", 13),
            text_color=COLORS["text_secondary"], justify="center"
        ).grid(row=1, column=0, columnspan=7, pady=70, sticky="ew")

    def _reset_table(self) -> None:
        for w in self.table_scroll.winfo_children():
            w.destroy()
        self._draw_table_header()
        self._draw_placeholder()

    # ── Calculation ───────────────────────────────────────────────────────────
    def _on_calculate(self) -> None:
        self._close_all_dropdowns()
        to_fetch: list[dict] = []

        for row in self.ingredient_rows:
            name = row["ing_entry"].get().strip()
            qty  = row["qty_entry"].get().strip()

            if not name and not qty:
                continue
            if not name:
                self._reset_table()
                self._set_status("⚠  Inserisci il nome di tutti gli ingredienti.",
                                 COLORS["warning"])
                return
            if not qty:
                self._reset_table()
                self._set_status(f"⚠  Quantità mancante per «{name}».", COLORS["warning"])
                return
            try:
                qty_val = float(qty.replace(",", "."))
            except ValueError:
                self._reset_table()
                self._set_status(f"⚠  Quantità non valida per «{name}».", COLORS["warning"])
                return
            if qty_val <= 0:
                self._reset_table()
                self._set_status(f"⚠  La quantità di «{name}» deve essere > 0.",
                                 COLORS["warning"])
                return
            if not row["fdc_id"]:
                self._reset_table()
                self._set_status(
                    f"⚠  Seleziona «{name}» dall'elenco suggerimenti.", COLORS["warning"])
                return

            to_fetch.append({"name": name, "quantity": qty_val, "fdc_id": row["fdc_id"]})

        if not to_fetch:
            self._reset_table()
            self._set_status("⚠  Inserisci almeno un ingrediente.", COLORS["warning"])
            return

        self.calc_btn.configure(state="disabled", text="⏳  Recupero dati…")
        self._set_status("Connessione a USDA FoodData Central…", COLORS["warning"])
        threading.Thread(target=self._fetch_all, args=(to_fetch,), daemon=True).start()

    def _fetch_all(self, to_fetch: list[dict]) -> None:
        results, errors = [], []
        for item in to_fetch:
            try:
                nuts = usda_nutrients(item["fdc_id"], item["quantity"])
                results.append({"name": item["name"], "quantity": item["quantity"], **nuts})
            except Exception:
                errors.append(item["name"])
        self.after(0, lambda: self._finish(results, errors))

    def _finish(self, results: list[dict], errors: list[str]) -> None:
        self.calc_btn.configure(state="normal", text="⚡  Calcola")
        if errors:
            self._reset_table()
            self._set_status(
                f"⚠  Errore nel recupero dati per: {', '.join(errors)}", COLORS["error"])
            return
        self._render_results(results)

    # ── Rendering ─────────────────────────────────────────────────────────────
    def _render_results(self, rows: list[dict]) -> None:
        for w in self.table_scroll.winfo_children():
            w.destroy()
        self._draw_table_header()

        totals = {k: 0.0 for k in
                  ("quantity", "calories", "proteins", "lipids", "sodium", "fibers")}

        for i, row in enumerate(rows):
            bg = COLORS["bg_row_even"] if i % 2 == 0 else COLORS["bg_row_odd"]
            for key in totals:
                totals[key] += row[key]

            vals = [
                row["name"],               fmt_qty(row["quantity"]),
                f"{row['calories']:.2f}",  f"{row['proteins']:.2f}",
                f"{row['lipids']:.2f}",    f"{row['sodium']:.2f}",
                f"{row['fibers']:.2f}",
            ]
            for col, val in enumerate(vals):
                ctk.CTkLabel(
                    self.table_scroll, text=val,
                    font=ctk.CTkFont("Segoe UI", 12),
                    fg_color=bg, text_color=COLORS["text_primary"],
                    corner_radius=0,
                    anchor="w" if col == 0 else "center",
                    justify="left" if col == 0 else "center"
                ).grid(row=i + 1, column=col, padx=1, pady=1,
                       sticky="ew", ipadx=8, ipady=5)

        # Ceil each total
        for key in totals:
            totals[key] = math.ceil(totals[key] * 100) / 100

        tv = [
            "TOTALE",                    fmt_qty(totals["quantity"]),
            f"{totals['calories']:.2f}", f"{totals['proteins']:.2f}",
            f"{totals['lipids']:.2f}",   f"{totals['sodium']:.2f}",
            f"{totals['fibers']:.2f}",
        ]
        for col, val in enumerate(tv):
            ctk.CTkLabel(
                self.table_scroll, text=val,
                font=ctk.CTkFont("Segoe UI", 13, "bold"),
                fg_color=COLORS["bg_total"], text_color="white",
                corner_radius=4,
                anchor="w" if col == 0 else "center",
                justify="left" if col == 0 else "center"
            ).grid(row=len(rows) + 1, column=col, padx=1, pady=(5, 2),
                   sticky="ew", ipadx=8, ipady=7)

        n = len(rows)
        self._set_status(
            f"✅  {n} ingredient{'e' if n == 1 else 'i'} calcolat{'o' if n == 1 else 'i'}.",
            COLORS["success"]
        )

    # ── Helpers ───────────────────────────────────────────────────────────────
    def _set_status(self, text: str, color: str) -> None:
        self.status_lbl.configure(text=text, text_color=color)
