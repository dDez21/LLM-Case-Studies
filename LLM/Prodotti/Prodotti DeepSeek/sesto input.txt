class SuggestionPopup(ctk.CTkToplevel):
    """Finestra popup per i suggerimenti dell'autocompletamento."""
    def __init__(self, parent, entry, commit_callback):
        super().__init__(parent)
        self.entry = entry
        self.commit_callback = commit_callback
        self.overrideredirect(True)  # nessuna barra del titolo
        self.attributes("-topmost", True)

        self.configure(fg_color="#2b2b2b")
        self.grid_columnconfigure(0, weight=1)

        # Area scorrevole con i suggerimenti
        self.scroll = ctk.CTkScrollableFrame(self, width=300, height=150)
        self.scroll.grid(row=0, column=0, sticky="nsew")

        self.suggestions = []  # memorizza i dati (fdcId, description)

    def set_suggestions(self, items):
        """Popola il popup con i suggerimenti e lo posiziona sotto l'entry."""
        # Pulisce i widget precedenti
        for w in self.scroll.winfo_children():
            w.destroy()
        self.suggestions = items

        if not items:
            self.withdraw()
            return

        for idx, item in enumerate(items):
            btn = ctk.CTkButton(
                self.scroll,
                text=item["description"],
                anchor="w",
                fg_color="transparent",
                hover_color="#3a3a3a",
                corner_radius=0,
                command=lambda i=idx: self.select(i)
            )
            btn.pack(fill="x", padx=2, pady=1)

        # Posizionamento
        x = self.entry.winfo_rootx()
        y = self.entry.winfo_rooty() + self.entry.winfo_height() + 2
        self.geometry(f"+{x}+{y}")
        self.deiconify()

    def select(self, idx):
        """Seleziona un suggerimento e richiama la callback."""
        item = self.suggestions[idx]
        self.commit_callback(item["fdcId"], item["description"])
        self.withdraw()


class NutritionCalculator(ctk.CTk):
    def __init__(self):
        super().__init__()
        self.title("Calcolatore Valori Nutrizionali")
        self.geometry("890x640")
        self.resizable(False, False)

        ctk.set_appearance_mode("dark")
        ctk.set_default_color_theme("dark-blue")

        self.left_frame = ctk.CTkFrame(self, width=450, height=600)
        self.left_frame.pack(side="left", fill="y", expand=False, padx=10, pady=10)
        self.left_frame.pack_propagate(False)

        self.right_frame = ctk.CTkFrame(self, width=400, height=600)
        self.right_frame.pack(side="right", fill="y", expand=False, padx=10, pady=10)
        self.right_frame.pack_propagate(False)

        # Oggetti per l'autocompletamento
        self.active_popup = None
        self.search_timer = None
        self.current_entry = None
        self.current_row = None

        self.build_left_panel()
        self.build_right_panel()

    # ------------------------------------------------------------
    # COSTRUZIONE INTERFACCIA
    # ------------------------------------------------------------
    def build_left_panel(self):
        ctk.CTkLabel(self.left_frame, text="Inserisci gli alimenti (max 15)",
                     font=("Arial", 16, "bold")).pack(pady=(10, 5))

        self.scroll_frame = ctk.CTkScrollableFrame(self.left_frame, width=420, height=400)
        self.scroll_frame.pack(fill="both", expand=True, padx=5, pady=5)

        self.ingredient_entries = []
        self.quantity_entries = []
        self.fdc_ids = []           # memorizza l'fdcId selezionato per ogni riga
        self.row_frames = []
        self.number_labels = []

        bottom_frame = ctk.CTkFrame(self.scroll_frame, fg_color="transparent")
        bottom_frame.pack(side="bottom", fill="x", padx=0, pady=(5, 10))

        self.add_button = ctk.CTkButton(
            bottom_frame, text="+ Aggiungi alimento",
            command=self.add_ingredient_row, width=120, height=28,
            font=("Arial", 11), fg_color="#2e8b57", hover_color="#1f6d3d"
        )
        self.add_button.pack(expand=True)

        self.calc_button = ctk.CTkButton(
            self.left_frame, text="Calcola valori nutrizionali",
            command=self.calculate, height=38,
            font=("Arial", 14, "bold"), fg_color="#1f538d", hover_color="#14375e"
        )
        self.calc_button.pack(pady=(5, 10))

    def add_ingredient_row(self):
        if len(self.ingredient_entries) >= 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")
            return

        row_frame = ctk.CTkFrame(self.scroll_frame)
        row_frame.pack(fill="x", padx=5, pady=2, before=self.add_button.master)

        num = len(self.ingredient_entries) + 1
        num_label = ctk.CTkLabel(row_frame, text=f"{num}.", width=30)
        num_label.pack(side="left", padx=5)

        ing_entry = ctk.CTkEntry(row_frame, placeholder_text="Cerca alimento...", width=180)
        ing_entry.pack(side="left", padx=5)

        qty_entry = ctk.CTkEntry(row_frame, placeholder_text="Quantità (g)", width=80)
        qty_entry.pack(side="left", padx=5)

        remove_btn = ctk.CTkButton(
            row_frame, text="🗑️", width=30, height=30,
            fg_color="#a94442", hover_color="#843534",
            command=lambda f=row_frame: self.remove_row(f)
        )
        remove_btn.pack(side="right", padx=5)

        self.ingredient_entries.append(ing_entry)
        self.quantity_entries.append(qty_entry)
        self.fdc_ids.append(None)         # nessun alimento selezionato
        self.row_frames.append(row_frame)
        self.number_labels.append(num_label)

        # Associa gli eventi per l'autocompletamento
        row_idx = len(self.ingredient_entries) - 1
        ing_entry.bind("<KeyRelease>", lambda e, idx=row_idx: self.on_key_release(idx))
        ing_entry.bind("<FocusOut>", lambda e: self.hide_popup())

        if len(self.ingredient_entries) == 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")

    def remove_row(self, frame):
        try:
            idx = self.row_frames.index(frame)
        except ValueError:
            return

        del self.ingredient_entries[idx]
        del self.quantity_entries[idx]
        del self.fdc_ids[idx]
        del self.row_frames[idx]
        del self.number_labels[idx]
        frame.destroy()

        for i in range(idx, len(self.number_labels)):
            self.number_labels[i].configure(text=f"{i+1}.")

        if len(self.ingredient_entries) < 15:
            self.add_button.configure(state="normal", text="+ Aggiungi alimento")

    # ------------------------------------------------------------
    # AUTOCOMPLETAMENTO
    # ------------------------------------------------------------
    def on_key_release(self, idx):
        """Avvia la ricerca con un debounce di 500 ms."""
        if self.search_timer is not None:
            self.after_cancel(self.search_timer)

        self.current_row = idx
        self.current_entry = self.ingredient_entries[idx]

        text = self.current_entry.get().strip()
        if len(text) < 2:
            self.hide_popup()
            return

        self.search_timer = self.after(500, lambda: self.search_food(text))

    def search_food(self, query):
        """Esegue la ricerca USDA in un thread separato."""
        def worker():
            try:
                params = {
                    "query": query,
                    "api_key": USDA_API_KEY,
                    "pageSize": 25,           # risultati sufficienti per filtrare
                    "dataType": ["Foundation", "SR Legacy", "Survey (FNDDS)", "Branded"]
                }
                resp = requests.get(USDA_SEARCH_URL, params=params, timeout=10)
                resp.raise_for_status()
                data = resp.json()
                foods = data.get("foods", [])

                # Ordina secondo i criteri richiesti
                sorted_foods = self.sort_results(foods)

                # Prepara i suggerimenti (massimo 10)
                suggestions = []
                for f in sorted_foods[:10]:
                    suggestions.append({
                        "fdcId": f["fdcId"],
                        "description": f["description"]
                    })

                self.after(0, lambda: self.show_suggestions(suggestions))
            except Exception as e:
                # In caso di errore, semplicemente non mostra suggerimenti
                print(f"Errore ricerca USDA: {e}")
                self.after(0, self.hide_popup)

        threading.Thread(target=worker, daemon=True).start()

    def sort_results(self, foods):
        """Ordina i risultati per dataType e poi per 'semplicità'."""
        # Calcola un punteggio per ogni alimento
        scored = []
        for f in foods:
            dt = f.get("dataType", "")
            base = DATA_TYPE_PRIORITY.get(dt, 99)

            # Penalità per descrizioni lunghe o contenenti parole complesse
            desc = f.get("description", "")
            words = desc.lower().split()
            word_penalty = max(0, len(words) - 3) * 0.5
            complexity = 0
            for kw in COMPLEX_KEYWORDS:
                if kw in desc.lower():
                    complexity += 1
            score = base + word_penalty + complexity * 2
            scored.append((score, f))

        scored.sort(key=lambda x: x[0])
        return [item[1] for item in scored]

    def show_suggestions(self, suggestions):
        """Crea o aggiorna il popup dei suggerimenti."""
        if not self.current_entry:
            return

        # Chiude il popup precedente se esiste
        self.hide_popup()

        if not suggestions:
            return

        self.active_popup = SuggestionPopup(
            self, self.current_entry,
            self.on_food_selected
        )
        self.active_popup.set_suggestions(suggestions)

    def hide_popup(self):
        if self.active_popup:
            self.active_popup.destroy()
            self.active_popup = None

    def on_food_selected(self, fdc_id, description):
        """Callback quando l'utente seleziona un alimento dal popup."""
        if self.current_row is not None and self.current_entry:
            self.current_entry.delete(0, "end")
            self.current_entry.insert(0, description)
            self.fdc_ids[self.current_row] = fdc_id
        self.hide_popup()

    # ------------------------------------------------------------
    # PANNELLO DESTRO (RISULTATI)
    # ------------------------------------------------------------
    def build_right_panel(self):
        ctk.CTkLabel(self.right_frame, text="Valori Nutrizionali Totali",
                     font=("Arial", 16, "bold")).pack(pady=(10, 15))

        table_box = ctk.CTkFrame(self.right_frame, fg_color="#2b2b2b",
                                 corner_radius=10, border_width=1,
                                 border_color="#4a4a4a")
        table_box.pack(fill="x", padx=15, pady=0, ipady=10)

        nutrients = [
            ("Calorie (kcal)", "calorie"),
            ("Proteine (g)",   "proteine"),
            ("Lipidi (g)",     "lipidi"),
            ("Sodio (mg)",     "sodio"),
            ("Fibre (g)",      "fibre")
        ]
        self.value_labels = {}
        for i, (name, key) in enumerate(nutrients):
            lbl_name = ctk.CTkLabel(table_box, text=name, font=("Arial", 14), anchor="w")
            lbl_name.grid(row=i, column=0, padx=(20, 10), pady=10, sticky="w")
            lbl_value = ctk.CTkLabel(table_box, text="0.00", font=("Arial", 16, "bold"), anchor="e")
            lbl_value.grid(row=i, column=1, padx=(10, 20), pady=10, sticky="e")
            self.value_labels[key] = lbl_value

        self.message_label = ctk.CTkLabel(self.right_frame, text="",
                                          text_color="#66bb6a", font=("Arial", 12))
        self.message_label.pack(pady=(15, 5))

    def reset_totals(self):
        for lbl in self.value_labels.values():
            lbl.configure(text="0.00")

    # ------------------------------------------------------------
    # CALCOLO
    # ------------------------------------------------------------
    def calculate(self):
        self.reset_totals()
        self.message_label.configure(text="")

        totals = {"calorie": 0.0, "proteine": 0.0, "lipidi": 0.0, "sodio": 0.0, "fibre": 0.0}
        found_any = False

        for idx, (ing_entry, qty_entry) in enumerate(zip(self.ingredient_entries, self.quantity_entries)):
            ing_text = ing_entry.get().strip()
            qty_str = qty_entry.get().strip()

            if not ing_text and not qty_str:
                continue

            if not ing_text or not qty_str:
                self.message_label.configure(
                    text="Completa entrambi i campi per ogni alimento inserito.",
                    text_color="#f0ad4e"
                )
                return

            try:
                qty = float(qty_str)
                if qty <= 0:
                    self.message_label.configure(
                        text="La quantità deve essere un numero positivo.",
                        text_color="#f0ad4e"
                    )
                    return
            except ValueError:
                self.message_label.configure(
                    text="Inserisci una quantità numerica valida (grammi).",
                    text_color="#f0ad4e"
                )
                return

            # Recupera l'fdcId selezionato
            fdc_id = self.fdc_ids[idx]
            if fdc_id is None:
                self.message_label.configure(
                    text=f"Seleziona un alimento valido dalla lista per '{ing_text}'.",
                    text_color="#f0ad4e"
                )
                return

            # Ottiene i nutrienti per 100g dall'API
            nutrients = self.fetch_nutrients(fdc_id)
            if nutrients is None:
                self.message_label.configure(
                    text=f"Impossibile recuperare i dati nutrizionali per '{ing_text}'.",
                    text_color="#f0ad4e"
                )
                return

            factor = qty / 100.0
            for key in totals:
                totals[key] += nutrients.get(key, 0.0) * factor

            found_any = True

        if not found_any:
            self.message_label.configure(
                text="Inserisci almeno un alimento per eseguire il calcolo.",
                text_color="#f0ad4e"
            )
            return

        # Arrotondamento per eccesso ai centesimi
        for key in totals:
            totals[key] = math.ceil(totals[key] * 100) / 100

        for key, lbl in self.value_labels.items():
            lbl.configure(text=f"{totals[key]:.2f}")

        self.message_label.configure(
            text="✓ Calcolo completato con successo.",
            text_color="#66bb6a"
        )

    def fetch_nutrients(self, fdc_id):
        """Recupera i nutrienti per 100g dell'alimento specificato."""
        try:
            url = USDA_FOOD_URL.format(fdc_id)
            params = {"api_key": USDA_API_KEY}
            resp = requests.get(url, params=params, timeout=10)
            resp.raise_for_status()
            data = resp.json()

            # Mappa id nutrienti USDA -> nostre chiavi
            # Energy (kcal) = 1008, Protein = 1003, Total lipid = 1004,
            # Sodium = 1093, Fiber = 1079
            target_nutrients = {
                1008: "calorie",
                1003: "proteine",
                1004: "lipidi",
                1093: "sodio",
                1079: "fibre"
            }
            result = {"calorie": 0.0, "proteine": 0.0, "lipidi": 0.0, "sodio": 0.0, "fibre": 0.0}

            for nut in data.get("foodNutrients", []):
                nut_id = nut.get("nutrientId")
                if nut_id in target_nutrients:
                    key = target_nutrients[nut_id]
                    # Il valore è per 100g se disponibile, altrimenti 0
                    value = nut.get("value", 0.0)
                    result[key] = float(value)

            return result
        except Exception as e:
            print(f"Errore nel recupero nutrienti per fdcId {fdc_id}: {e}")
            return None
