NUTRIENTS = ("Calorie", "Proteine", "Lipidi", "Sodio", "Fibre")
MAX_INGREDIENTS = 15


class NutritionCalculator(ctk.CTk):
    def __init__(self):
        super().__init__()
        self.title("Calcolatore nutrizionale")
        self.geometry("980x620")
        self.minsize(840, 540)

        ctk.set_appearance_mode("light")
        ctk.set_default_color_theme("green")

        self.ingredient_entries = []
        self.quantity_entries = []
        self.remove_buttons = []
        self.result_vars = {nutrient: ctk.StringVar(value="0.00") for nutrient in NUTRIENTS}

        self._build_layout()

    def _build_layout(self):
        self.grid_columnconfigure(0, weight=1, uniform="main")
        self.grid_columnconfigure(1, weight=1, uniform="main")
        self.grid_rowconfigure(0, weight=1)

        input_frame = ctk.CTkFrame(self, corner_radius=8)
        input_frame.grid(row=0, column=0, padx=(18, 9), pady=18, sticky="nsew")
        input_frame.grid_columnconfigure((0, 1), weight=1)
        input_frame.grid_rowconfigure(2, weight=1)

        results_frame = ctk.CTkFrame(self, corner_radius=8)
        results_frame.grid(row=0, column=1, padx=(9, 18), pady=18, sticky="nsew")
        results_frame.grid_columnconfigure(0, weight=1)
        results_frame.grid_rowconfigure(1, weight=1)

        ctk.CTkLabel(
            results_frame,
            text="Totale valori nutrizionali",
            font=ctk.CTkFont(size=22, weight="bold"),
        ).grid(row=0, column=0, padx=20, pady=(20, 12), sticky="w")

        self._build_input_area(input_frame)
        self._build_results_table(results_frame)

    def _build_results_table(self, parent):
        table_frame = ctk.CTkFrame(parent, fg_color="transparent")
        table_frame.grid(row=1, column=0, padx=20, pady=(0, 20), sticky="nsew")
        table_frame.grid_columnconfigure(0, weight=1)
        table_frame.grid_columnconfigure(1, weight=1)

        header_style = ctk.CTkFont(size=15, weight="bold")
        ctk.CTkLabel(table_frame, text="Valore", font=header_style).grid(
            row=0, column=0, padx=10, pady=10, sticky="w"
        )
        ctk.CTkLabel(table_frame, text="Totale", font=header_style).grid(
            row=0, column=1, padx=10, pady=10, sticky="e"
        )

        for row, nutrient in enumerate(NUTRIENTS, start=1):
            unit = "kcal" if nutrient == "Calorie" else "mg" if nutrient == "Sodio" else "g"

            ctk.CTkLabel(table_frame, text=nutrient, font=ctk.CTkFont(size=15)).grid(
                row=row, column=0, padx=10, pady=12, sticky="w"
            )
            ctk.CTkLabel(
                table_frame,
                textvariable=self.result_vars[nutrient],
                font=ctk.CTkFont(size=15, weight="bold"),
            ).grid(row=row, column=1, padx=(10, 4), pady=12, sticky="e")
            ctk.CTkLabel(table_frame, text=unit, font=ctk.CTkFont(size=13)).grid(
                row=row, column=2, padx=(0, 10), pady=12, sticky="w"
            )

        ctk.CTkLabel(
            parent,
            text="Alimenti disponibili: " + ", ".join(sorted(NUTRITION_PER_100G)),
            wraplength=380,
            justify="left",
            text_color="#4b5563",
        ).grid(row=2, column=0, padx=20, pady=(0, 20), sticky="w")

    def _build_input_area(self, parent):
        ctk.CTkLabel(
            parent,
            text="Ingredienti",
            font=ctk.CTkFont(size=22, weight="bold"),
        ).grid(row=0, column=0, columnspan=2, padx=20, pady=(20, 12), sticky="w")

        self.rows_frame = ctk.CTkScrollableFrame(parent, fg_color="transparent")
        self.rows_frame.grid(row=2, column=0, columnspan=2, padx=12, pady=(0, 10), sticky="nsew")
        self.rows_frame.grid_columnconfigure(0, weight=1)
        self.rows_frame.grid_columnconfigure(1, weight=1)
        self.rows_frame.grid_columnconfigure(2, weight=0)

        ctk.CTkLabel(self.rows_frame, text="Ingrediente", font=ctk.CTkFont(weight="bold")).grid(
            row=0, column=0, padx=(8, 8), pady=(0, 8), sticky="w"
        )
        ctk.CTkLabel(self.rows_frame, text="Quantità (g)", font=ctk.CTkFont(weight="bold")).grid(
            row=0, column=1, padx=(8, 8), pady=(0, 8), sticky="w"
        )

        self.add_button = ctk.CTkButton(
            self.rows_frame,
            text="+ Aggiungi alimento",
            command=self.add_ingredient_row,
            width=150,
            height=30,
            font=ctk.CTkFont(size=13),
        )

        ctk.CTkButton(
            parent,
            text="Calcola valori nutrizionali",
            command=self.calculate_totals,
            height=42,
            font=ctk.CTkFont(size=15, weight="bold"),
        ).grid(row=3, column=0, columnspan=2, padx=20, pady=(0, 20), sticky="ew")

        self.add_ingredient_row()

    def add_ingredient_row(self):
        if len(self.ingredient_entries) >= MAX_INGREDIENTS:
            return

        row = len(self.ingredient_entries)
        ingredient_entry = ctk.CTkEntry(self.rows_frame, placeholder_text=f"Ingrediente {row}")

        quantity_entry = ctk.CTkEntry(self.rows_frame, placeholder_text="0")

        remove_button = ctk.CTkButton(
            self.rows_frame,
            text="Rimuovi",
            width=72,
            height=28,
            fg_color="#b91c1c",
            hover_color="#991b1b",
            command=lambda entry=ingredient_entry: self.remove_ingredient_row(entry),
        )

        self.ingredient_entries.append(ingredient_entry)
        self.quantity_entries.append(quantity_entry)
        self.remove_buttons.append(remove_button)

        self._refresh_ingredient_rows()

    def remove_ingredient_row(self, ingredient_entry):
        if ingredient_entry not in self.ingredient_entries:
            return

        index = self.ingredient_entries.index(ingredient_entry)
        widgets = (
            self.ingredient_entries.pop(index),
            self.quantity_entries.pop(index),
            self.remove_buttons.pop(index),
        )

        for widget in widgets:
            widget.destroy()

        if not self.ingredient_entries:
            self.add_ingredient_row()
            return

        self._refresh_ingredient_rows()

    def _refresh_ingredient_rows(self):
        self.add_button.grid_forget()

        for index, (ingredient_entry, quantity_entry, remove_button) in enumerate(
            zip(self.ingredient_entries, self.quantity_entries, self.remove_buttons),
            start=1,
        ):
            ingredient_entry.configure(placeholder_text=f"Ingrediente {index}")
            ingredient_entry.grid(row=index, column=0, padx=(8, 8), pady=5, sticky="ew")
            quantity_entry.grid(row=index, column=1, padx=(8, 8), pady=5, sticky="ew")
            remove_button.grid(row=index, column=2, padx=(4, 8), pady=5, sticky="e")

        add_row = len(self.ingredient_entries) + 1
        self.add_button.grid(row=add_row, column=0, columnspan=3, padx=(8, 8), pady=(6, 4), sticky="w")

        if len(self.ingredient_entries) == MAX_INGREDIENTS:
            self.add_button.configure(state="disabled", text="Limite di 15 alimenti raggiunto")
        else:
            self.add_button.configure(state="normal", text="+ Aggiungi alimento")

    def calculate_totals(self):
        totals = {nutrient: 0.0 for nutrient in NUTRIENTS}
        unknown_ingredients = []

        for ingredient_entry, quantity_entry in zip(self.ingredient_entries, self.quantity_entries):
            ingredient = ingredient_entry.get().strip().lower()
            quantity_text = quantity_entry.get().strip().replace(",", ".")

            if not ingredient and not quantity_text:
                continue

            if not ingredient or not quantity_text:
                messagebox.showerror("Dati incompleti", "Compila sia Ingrediente sia Quantità.")
                return

            try:
                quantity = float(quantity_text)
            except ValueError:
                messagebox.showerror("Quantità non valida", f"La quantità di '{ingredient}' non è un numero.")
                return

            if quantity < 0:
                messagebox.showerror("Quantità non valida", "Inserisci quantità maggiori o uguali a zero.")
                return

            nutrition = NUTRITION_PER_100G.get(ingredient)
            if nutrition is None:
                unknown_ingredients.append(ingredient)
                continue

            multiplier = quantity / 100
            for nutrient in NUTRIENTS:
                totals[nutrient] += nutrition[nutrient] * multiplier

        for nutrient, total in totals.items():
            self.result_vars[nutrient].set(f"{self._round_up(total):.2f}")

        if unknown_ingredients:
            messagebox.showwarning(
                "Ingredienti non trovati",
                "Non ho trovato questi alimenti nel dizionario locale: "
                + ", ".join(sorted(set(unknown_ingredients)))
                + ". Puoi aggiungerli in NUTRITION_PER_100G.",
            )

    @staticmethod
    def _round_up(value):
        return math.ceil((value - 1e-9) * 100) / 100


if __name__ == "__main__":
    app = NutritionCalculator()
    app.mainloop()
