    def add_ingredient_row(self):
        if len(self.ingredient_entries) >= MAX_INGREDIENTS:
            return

        row = len(self.ingredient_entries)
        ingredient_entry = ctk.CTkEntry(
            self.rows_frame,
            placeholder_text=f"Ingrediente {row}",
            fg_color=COLORS["input"],
            border_color=COLORS["border"],
            text_color=COLORS["text"],
            placeholder_text_color=COLORS["muted"],
        )

        quantity_entry = ctk.CTkEntry(
            self.rows_frame,
            placeholder_text="0",
            fg_color=COLORS["input"],
            border_color=COLORS["border"],
            text_color=COLORS["text"],
            placeholder_text_color=COLORS["muted"],
        )

        remove_button = ctk.CTkButton(
            self.rows_frame,
            text="\U0001F5D1",
            width=34,
            height=30,
            corner_radius=6,
            fg_color="#b91c1c",
            hover_color="#991b1b",
            font=ctk.CTkFont(size=15),
            command=lambda entry=ingredient_entry: self.remove_ingredient_row(entry),
        )

        self.ingredient_entries.append(ingredient_entry)
        self.quantity_entries.append(quantity_entry)
        self.remove_buttons.append(remove_button)

        self._refresh_ingredient_rows()

    def remove_ingredient_row(self, ingredient_entry):
        if ingredient_entry not in self.ingredient_entries:
            return

        index = self.ingredient_entries.index(ingredient_entry)
        widgets = (
            self.ingredient_entries.pop(index),
            self.quantity_entries.pop(index),
            self.remove_buttons.pop(index),
        )

        for widget in widgets:
            widget.destroy()

        if not self.ingredient_entries:
            self.add_ingredient_row()
            return

        self._refresh_ingredient_rows()

    def _refresh_ingredient_rows(self):
        self.add_button.grid_forget()

        for index, (ingredient_entry, quantity_entry, remove_button) in enumerate(
            zip(self.ingredient_entries, self.quantity_entries, self.remove_buttons),
            start=1,
        ):
            ingredient_entry.configure(placeholder_text=f"Ingrediente {index}")
            ingredient_entry.grid(row=index, column=0, padx=(8, 8), pady=5, sticky="ew")
            quantity_entry.grid(row=index, column=1, padx=(8, 8), pady=5, sticky="ew")
            remove_button.grid(row=index, column=2, padx=(4, 8), pady=5, sticky="e")

        add_row = len(self.ingredient_entries) + 1
        self.add_button.grid(row=add_row, column=0, columnspan=3, padx=(8, 8), pady=(6, 4))

        if len(self.ingredient_entries) == MAX_INGREDIENTS:
            self.add_button.configure(state="disabled", text="Limite 15")
        else:
            self.add_button.configure(state="normal", text="+ Aggiungi alimento")

    def calculate_totals(self):
        totals = {nutrient: 0.0 for nutrient in NUTRIENTS}
        unknown_ingredients = []
        api_errors = []

        self.status_var.set("Calcolo in corso: interrogazione USDA FoodData Central API...")
        self.update_idletasks()

        for ingredient_entry, quantity_entry in zip(self.ingredient_entries, self.quantity_entries):
            ingredient = ingredient_entry.get().strip().lower()
            quantity_text = quantity_entry.get().strip().replace(",", ".")

            if not ingredient and not quantity_text:
                continue

            if not ingredient or not quantity_text:
                messagebox.showerror("Dati incompleti", "Compila sia Ingrediente sia Quantità.")
                self.status_var.set("Calcolo interrotto: dati incompleti.")
                return

            try:
                quantity = float(quantity_text)
            except ValueError:
                messagebox.showerror("Quantità non valida", f"La quantità di '{ingredient}' non è un numero.")
                self.status_var.set("Calcolo interrotto: quantita non numerica.")
                return

            if quantity < 0:
                messagebox.showerror("Quantità non valida", "Inserisci quantità maggiori o uguali a zero.")
                self.status_var.set("Calcolo interrotto: quantita non valida.")
                return

            try:
                nutrition = self.fetch_usda_nutrition(ingredient)
            except RuntimeError as error:
                api_errors.append(str(error))
                continue

            if not nutrition:
                unknown_ingredients.append(ingredient)
                continue

            multiplier = quantity / 100
            for nutrient in NUTRIENTS:
                totals[nutrient] += nutrition.get(nutrient, 0.0) * multiplier

        for nutrient, total in totals.items():
            self.result_vars[nutrient].set(f"{self._round_up(total):.2f}")

        self.status_var.set("Fonte dati: USDA FoodData Central API.")

        if api_errors:
            messagebox.showerror("Errore USDA FoodData Central", "\n".join(api_errors[:3]))
            self.status_var.set("Errore durante la richiesta a USDA FoodData Central API.")
            return

        if unknown_ingredients:
            messagebox.showwarning(
                "Ingredienti non trovati",
                "Non ho trovato questi alimenti tramite USDA FoodData Central API: "
                + ", ".join(sorted(set(unknown_ingredients)))
            )

    def fetch_usda_nutrition(self, ingredient):
        if ingredient in self.usda_cache:
            return self.usda_cache[ingredient]

        query = urllib.parse.urlencode({"api_key": USDA_API_KEY})
        payload = json.dumps(
            {
                "query": ingredient,
                "pageSize": 1,
                "requireAllWords": False,
            }
        ).encode("utf-8")
        request = urllib.request.Request(
            f"{USDA_SEARCH_URL}?{query}",
            data=payload,
            headers={"Content-Type": "application/json"},
            method="POST",
        )

        try:
            with urllib.request.urlopen(request, timeout=15) as response:
                data = json.loads(response.read().decode("utf-8"))
        except urllib.error.HTTPError as error:
            raise RuntimeError(f"USDA API ha risposto con errore HTTP {error.code}.") from error
        except urllib.error.URLError as error:
            raise RuntimeError(f"Impossibile collegarsi alla USDA API: {error.reason}.") from error
        except TimeoutError as error:
            raise RuntimeError("La richiesta alla USDA API ha superato il tempo massimo.") from error

        foods = data.get("foods", [])
        if not foods:
            self.usda_cache[ingredient] = None
            return None

        nutrition = self.extract_nutrients_from_usda_food(foods[0])
        self.usda_cache[ingredient] = nutrition
        return nutrition

    @staticmethod
    def extract_nutrients_from_usda_food(food):
        nutrition = {nutrient: 0.0 for nutrient in NUTRIENTS}

        for food_nutrient in food.get("foodNutrients", []):
            nutrient_id = str(food_nutrient.get("nutrientId", "")).strip()
            nutrient_value = food_nutrient.get("value")

            for nutrient_name, usda_id in USDA_NUTRIENTS.items():
                if nutrient_id == usda_id and nutrient_value is not None:
                    nutrition[nutrient_name] = float(nutrient_value)

        return nutrition

    @staticmethod
    def _round_up(value):
        return math.ceil((value - 1e-9) * 100) / 100
