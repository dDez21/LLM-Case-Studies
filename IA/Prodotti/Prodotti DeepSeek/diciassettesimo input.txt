class SuggestionPopup(ctk.CTkToplevel):
    def __init__(self, parent, entry, fdc_ids, row_idx):
        super().__init__(parent)
        self.entry = entry
        self.fdc_ids = fdc_ids
        self.row_idx = row_idx
        self.overrideredirect(True)
        self.attributes("-topmost", True)
        self.configure(fg_color="#2b2b2b")
        self.grid_columnconfigure(0, weight=1)

        self.scroll = ctk.CTkScrollableFrame(self, width=300, height=150)
        self.scroll.grid(row=0, column=0, sticky="nsew")
        self.suggestions = []

    def set_suggestions(self, items):
        for w in self.scroll.winfo_children():
            w.destroy()
        self.suggestions = items
        if not items:
            self.withdraw()
            return

        for idx, item in enumerate(items):
            btn = ctk.CTkButton(
                self.scroll,
                text=item["description"],
                anchor="w",
                fg_color="transparent",
                hover_color="#3a3a3a",
                corner_radius=0,
                command=lambda i=idx: self.select(i)
            )
            btn.pack(fill="x", padx=2, pady=1)

        x = self.entry.winfo_rootx()
        y = self.entry.winfo_rooty() + self.entry.winfo_height() + 2
        self.geometry(f"+{x}+{y}")
        self.deiconify()

    def select(self, idx):
        item = self.suggestions[idx]
        self.entry.delete(0, "end")
        self.entry.insert(0, item["description"])
        self.fdc_ids[self.row_idx] = item["fdcId"]
        self.destroy()

    def add_ingredient_row(self):
        if not self.api_available:
            self.search_status.configure(text="API non disponibile", text_color="#f0ad4e")
            return
        if len(self.ingredient_entries) >= 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")
            return

        row_frame = ctk.CTkFrame(self.scroll_frame)
        row_frame.pack(fill="x", padx=5, pady=2, before=self.add_button.master)

        num = len(self.ingredient_entries) + 1
        num_label = ctk.CTkLabel(row_frame, text=f"{num}.", width=30)
        num_label.pack(side="left", padx=5)

        ing_entry = ctk.CTkEntry(row_frame, placeholder_text="Cerca alimento...", width=180)
        ing_entry.pack(side="left", padx=5)

        qty_entry = ctk.CTkEntry(row_frame, placeholder_text="Quantità (g)", width=80)
        qty_entry.pack(side="left", padx=5)

        remove_btn = ctk.CTkButton(
            row_frame, text="🗑️", width=30, height=30,
            fg_color="#a94442", hover_color="#843534",
            command=lambda f=row_frame: self.remove_row(f)
        )
        remove_btn.pack(side="right", padx=5)

        self.ingredient_entries.append(ing_entry)
        self.quantity_entries.append(qty_entry)
        self.fdc_ids.append(None)
        self.row_frames.append(row_frame)
        self.number_labels.append(num_label)

        row_idx = len(self.ingredient_entries) - 1
        ing_entry.bind("<KeyRelease>", lambda e, idx=row_idx: self.on_key_release(idx))
        # Nasconde il popup quando l'entry perde il focus (con ritardo)
        ing_entry.bind("<FocusOut>", lambda e, idx=row_idx: self.schedule_hide_popup())

        if len(self.ingredient_entries) == 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")

    def remove_row(self, frame):
        try:
            idx = self.row_frames.index(frame)
        except ValueError:
            return

        # Chiude il popup se appartiene a questa riga
        if self.active_popup and self.active_popup.row_idx == idx:
            self.hide_popup()

        del self.ingredient_entries[idx]
        del self.quantity_entries[idx]
        del self.fdc_ids[idx]
        del self.row_frames[idx]
        del self.number_labels[idx]
        frame.destroy()

        for i in range(idx, len(self.number_labels)):
            self.number_labels[i].configure(text=f"{i+1}.")

        if len(self.ingredient_entries) < 15:
            self.add_button.configure(state="normal", text="+ Aggiungi alimento")

    # ------------------------------------------------------------
    # AUTOCOMPLETAMENTO (ricerca robustissima)
    # ------------------------------------------------------------
    def schedule_hide_popup(self):
        """Nasconde il popup dopo un breve ritardo, cancellabile dalla selezione."""
        if self.popup_hide_timer is not None:
            self.after_cancel(self.popup_hide_timer)
        self.popup_hide_timer = self.after(200, self.hide_popup)

    def on_key_release(self, idx):
        if self.search_timer is not None:
            self.after_cancel(self.search_timer)

        entry = self.ingredient_entries[idx]
        text = entry.get().strip()
        if len(text) < 2:
            self.hide_popup()
            self.search_status.configure(text="")
            return

        if not self.api_available:
            self.search_status.configure(text="API non connessa", text_color="#f0ad4e")
            return

        self.search_status.configure(text="Ricerca in corso…")
        self.search_timer = self.after(500, lambda: self.search_food(text, idx, entry))

    def search_food(self, query, idx, entry):
        def worker():
            try:
                # Costruzione URL robustissima: ogni componente codificato singolarmente
                base_url = USDA_SEARCH_URL + "?query=" + urllib.parse.quote(query, safe='')
                base_url += "&api_key=" + urllib.parse.quote(USDA_API_KEY, safe='')
                base_url += "&pageSize=25"
                # DataType codificati senza caratteri speciali
                for dt in ["Foundation", "SR Legacy", "Survey (FNDDS)"]:
                    base_url += "&dataType=" + urllib.parse.quote(dt, safe='')
                url = base_url

                print("[RICERCA] URL:", url)

                resp = requests.get(url, timeout=10)
                try:
                    data = resp.json()
                except (json.JSONDecodeError, ValueError):
                    msg = f"Risposta API non valida (codice {resp.status_code}). Dettaglio: {resp.text[:200]}"
                    print("[RICERCA] Errore JSON:", msg)
                    self.after(0, lambda m=msg: self.show_api_error(m))
                    return

                if resp.status_code != 200:
                    error_detail = data.get("error", str(data))
                    msg = f"Errore API ({resp.status_code}): {error_detail}"
                    print("[RICERCA] Errore:", msg)
                    self.after(0, lambda m=msg: self.show_api_error(m))
                    return

                foods = data.get("foods", [])
                sorted_foods = self.sort_results(foods)
                suggestions = [
                    {"fdcId": f["fdcId"], "description": f["description"]}
                    for f in sorted_foods[:10]
                ]
                print(f"[RICERCA] Trovati {len(foods)} alimenti, mostrati {len(suggestions)}")
                self.after(0, lambda: self.show_suggestions(suggestions, idx, entry))
            except requests.RequestException as e:
                msg = f"Errore di rete: {str(e)}"
                print("[RICERCA] Eccezione:", msg)
                self.after(0, lambda m=msg: self.show_api_error(m))
            except Exception as e:
                msg = f"Errore imprevisto: {str(e)}"
                print("[RICERCA] Eccezione generica:", msg)
                self.after(0, lambda m=msg: self.show_api_error(m))

        threading.Thread(target=worker, daemon=True).start()

    def sort_results(self, foods):
        scored = []
        for f in foods:
            dt = f.get("dataType", "")
            base = DATA_TYPE_PRIORITY.get(dt, 99)
            desc = f.get("description", "")
            word_penalty = max(0, len(desc.split()) - 3) * 0.5
            complexity = sum(1 for kw in COMPLEX_KEYWORDS if kw in desc.lower())
            score = base + word_penalty + complexity * 2
            scored.append((score, f))
        scored.sort(key=lambda x: x[0])
        return [item[1] for item in scored]

    def show_suggestions(self, suggestions, idx, entry):
        self.search_status.configure(text="")
        # Cancella eventuale timer di chiusura prima di mostrare il nuovo popup
        if self.popup_hide_timer is not None:
            self.after_cancel(self.popup_hide_timer)
            self.popup_hide_timer = None
        self.hide_popup()
        if not suggestions:
            return
        self.active_popup = SuggestionPopup(self, entry, self.fdc_ids, idx)
        self.active_popup.set_suggestions(suggestions)

    def hide_popup(self):
        if self.active_popup:
            self.active_popup.destroy()
            self.active_popup = None
        if self.popup_hide_timer is not None:
            self.after_cancel(self.popup_hide_timer)
            self.popup_hide_timer = None

    def show_api_error(self, msg):
        self.search_status.configure(text="")
        self.hide_popup()
        self.message_label.configure(text=f"Errore API: {msg}", text_color="#f0ad4e")

    def reset_totals(self):
        for lbl in self.value_labels.values():
            lbl.configure(text="0.00")

    # ------------------------------------------------------------
    # CALCOLO (estrazione robusta dei nutrienti)
    # ------------------------------------------------------------
    def start_calculation(self):
        if not self.api_available:
            self.message_label.configure(text="API non disponibile. Controlla la chiave.", text_color="#f0ad4e")
            return
        if self.calc_thread and self.calc_thread.is_alive():
            return

        self.reset_totals()
        self.message_label.configure(text="Calcolo in corso…", text_color="#aaaaaa")
        self.calc_button.configure(state="disabled")
        self.calc_thread = threading.Thread(target=self.calculate_worker, daemon=True)
        self.calc_thread.start()

    def calculate_worker(self):
        totals = {"calorie": 0.0, "proteine": 0.0, "lipidi": 0.0, "sodio": 0.0, "fibre": 0.0}
        found_any = False
        error_message = None

        for idx, (ing_entry, qty_entry) in enumerate(zip(self.ingredient_entries, self.quantity_entries)):
            ing_text = ing_entry.get().strip()
            qty_str = qty_entry.get().strip()

            if not ing_text and not qty_str:
                continue

            if not ing_text or not qty_str:
                error_message = "Completa entrambi i campi per ogni alimento inserito."
                break

            try:
                qty = float(qty_str)
                if qty <= 0:
                    error_message = "La quantità deve essere un numero positivo."
                    break
            except ValueError:
                error_message = "Inserisci una quantità numerica valida (grammi)."
                break

            fdc_id = self.fdc_ids[idx]
            if fdc_id is None:
                error_message = f"Seleziona un alimento valido dalla lista per '{ing_text}'."
                break

            nutrients = self.fetch_nutrients(fdc_id)
            if nutrients is None:
                error_message = f"Impossibile recuperare i dati nutrizionali per '{ing_text}' (fdcId={fdc_id})."
                break

            print(f"[CALCOLO] {ing_text} (fdcId={fdc_id}): {nutrients}")
            factor = qty / 100.0
            for key in totals:
                totals[key] += nutrients.get(key, 0.0) * factor

            found_any = True

        if error_message:
            self.after(0, lambda m=error_message: self.finish_calculation(False, m, None))
            return
        if not found_any:
            self.after(0, lambda: self.finish_calculation(False, "Inserisci almeno un alimento per eseguire il calcolo.", None))
            return

        for key in totals:
            totals[key] = math.ceil(totals[key] * 100) / 100

        print("[CALCOLO] Totali finali:", totals)
        self.after(0, lambda t=totals: self.finish_calculation(True, None, t))

    def finish_calculation(self, success, message, totals):
        self.calc_button.configure(state="normal")
        if not success:
            self.reset_totals()
            self.message_label.configure(text=message, text_color="#f0ad4e")
            return
        for key, lbl in self.value_labels.items():
            lbl.configure(text=f"{totals[key]:.2f}")
        self.message_label.configure(text="✓ Calcolo completato con successo.", text_color="#66bb6a")

    def fetch_nutrients(self, fdc_id):
        """Recupera i nutrienti per 100g, gestendo la struttura annidata e labelNutrients."""
        try:
            url = USDA_FOOD_URL.format(fdc_id) + "?api_key=" + urllib.parse.quote(USDA_API_KEY, safe='')
            print("[FETCH NUTRIENTS] URL:", url)
            resp = requests.get(url, timeout=10)
            try:
                data = resp.json()
            except (json.JSONDecodeError, ValueError):
                print(f"[FETCH NUTRIENTS] Risposta non JSON per fdcId {fdc_id}: {resp.text[:200]}")
                return None

            if resp.status_code != 200:
                error = data.get("error", resp.text)
                print(f"[FETCH NUTRIENTS] Errore fetch nutrienti per {fdc_id}: {resp.status_code} {error}")
                return None

            # Insiemi di ID e nomi per ogni nutriente
            id_map = {
                "calorie": {1008, 2047, 2048},
                "proteine": {1003, 1162},
                "lipidi": {1004, 1258},
                "sodio": {1093, 1259},
                "fibre": {1079, 1268, 1292}
            }
            name_map = {
                "calorie": ["energy"],
                "proteine": ["protein"],
                "lipidi": ["fat", "lipid"],
                "sodio": ["sodium"],
                "fibre": ["fiber", "fibre"]
            }

            result = {"calorie": 0.0, "proteine": 0.0, "lipidi": 0.0, "sodio": 0.0, "fibre": 0.0}

            # --- 1. Estrazione da foodNutrients (struttura annidata) ---
            food_nutrients = data.get("foodNutrients", [])
            for nut in food_nutrients:
                # Supporta sia formato annidato che piatto
                nutrient_info = nut.get("nutrient", nut)
                nid = nutrient_info.get("id")
                name = (nutrient_info.get("name") or "").lower()
                amount = float(nut.get("amount", 0.0))

                # Controllo ID
                for key, ids in id_map.items():
                    if nid in ids:
                        result[key] = amount
                        print(f"  -> {key} (ID {nid}): {amount}")
                        break
                # Controllo nome (se ancora zero)
                for key, keywords in name_map.items():
                    if result[key] == 0.0:
                        if any(kw in name for kw in keywords):
                            result[key] = amount
                            print(f"  -> {key} (nome '{nutrient_info.get('name')}'): {amount}")
                            break

            # --- 2. Estrazione da labelNutrients (Branded) ---
            label_nutrients = data.get("labelNutrients", {})
            if label_nutrients:
                print("[FETCH NUTRIENTS] Trovato labelNutrients:", label_nutrients)
                for key, field_map in {
                    "calorie": ["calories", "energy"],
                    "proteine": ["protein"],
                    "lipidi": ["fat", "totalFat"],
                    "sodio": ["sodium"],
                    "fibre": ["fiber", "dietaryFiber"]
                }.items():
                    if result[key] == 0.0:
                        for field in field_map:
                            if field in label_nutrients:
                                val = float(label_nutrients[field].get("value", 0.0))
                                result[key] = val
                                print(f"  -> {key} (labelNutrients.{field}): {val}")
                                break

            # Se ancora zeri, stampa i nutrienti disponibili per debug
            missing = [k for k, v in result.items() if v == 0.0]
            if missing:
                print(f"[FETCH NUTRIENTS] Nutrienti mancanti: {missing}")
                if food_nutrients:
                    for n in food_nutrients[:5]:
                        print(f"   {n}")
                if label_nutrients:
                    print("   labelNutrients:", list(label_nutrients.keys())[:10])

            return result
        except Exception as e:
            print(f"[FETCH NUTRIENTS] Eccezione: {e}")
            return None
