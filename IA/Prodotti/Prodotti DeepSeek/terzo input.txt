class NutritionCalculator(ctk.CTk):
    def __init__(self):
        super().__init__()
        self.title("Calcolatore Valori Nutrizionali")
        self.geometry("950x650")
        ctk.set_appearance_mode("system")
        ctk.set_default_color_theme("blue")

        # Layout a due colonne
        self.left_frame = ctk.CTkFrame(self, width=450)
        self.left_frame.pack(side="left", fill="both", expand=True, padx=10, pady=10)

        self.right_frame = ctk.CTkFrame(self, width=400)
        self.right_frame.pack(side="right", fill="both", expand=True, padx=10, pady=10)

        self.build_left_panel()
        self.build_right_panel()

    def build_left_panel(self):
        """Pannello sinistro: titolo, area scorrevole, pulsante '+' in fondo e righe dinamiche."""
        # Titolo
        ctk.CTkLabel(
            self.left_frame,
            text="Inserisci gli alimenti (max 15)",
            font=("Arial", 16, "bold")
        ).pack(pady=(10, 5))

        # Contenitore scorrevole per le righe (inizialmente vuoto)
        self.scroll_frame = ctk.CTkScrollableFrame(self.left_frame, width=420, height=400)
        self.scroll_frame.pack(fill="both", expand=True, padx=5, pady=5)

        # Liste per memorizzare i widget di input e riferimenti alle righe
        self.ingredient_entries = []
        self.quantity_entries = []
        self.row_frames = []
        self.number_labels = []

        # Pulsante "+" posizionato in fondo all'area scorrevole (sempre visibile, sotto le righe)
        self.add_button = ctk.CTkButton(
            self.scroll_frame,
            text="+ Aggiungi alimento",
            command=self.add_ingredient_row,
            width=120,
            height=30,
            font=("Arial", 11)
        )
        self.add_button.pack(side="bottom", fill="x", padx=5, pady=(5, 10))

        # Pulsante per il calcolo (fuori dallo scroll)
        self.calc_button = ctk.CTkButton(
            self.left_frame,
            text="Calcola valori nutrizionali",
            command=self.calculate,
            height=40,
            font=("Arial", 14, "bold")
        )
        self.calc_button.pack(pady=(5, 10))

    def add_ingredient_row(self):
        """Aggiunge una nuova riga di input (fino a 15)."""
        if len(self.ingredient_entries) >= 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")
            return

        # Crea un frame per la riga
        row_frame = ctk.CTkFrame(self.scroll_frame)
        # Impacchetta sopra il pulsante "+" (che è in fondo) -> verrà inserita prima
        row_frame.pack(fill="x", padx=5, pady=2, before=self.add_button)

        # Numero progressivo
        num = len(self.ingredient_entries) + 1
        num_label = ctk.CTkLabel(row_frame, text=f"{num}.", width=30)
        num_label.pack(side="left", padx=5)

        # Campo ingrediente
        ing_entry = ctk.CTkEntry(row_frame, placeholder_text="Ingrediente", width=160)
        ing_entry.pack(side="left", padx=5)

        # Campo quantità (grammi)
        qty_entry = ctk.CTkEntry(row_frame, placeholder_text="Quantità (g)", width=80)
        qty_entry.pack(side="left", padx=5)

        # Pulsante rimuovi (X)
        remove_btn = ctk.CTkButton(
            row_frame,
            text="✕",
            width=25,
            height=25,
            fg_color="#d9534f",
            hover_color="#c9302c",
            command=lambda f=row_frame: self.remove_row(f)
        )
        remove_btn.pack(side="right", padx=5)

        # Salva i riferimenti
        self.ingredient_entries.append(ing_entry)
        self.quantity_entries.append(qty_entry)
        self.row_frames.append(row_frame)
        self.number_labels.append(num_label)

        # Se raggiunto il massimo, disabilita il pulsante
        if len(self.ingredient_entries) == 15:
            self.add_button.configure(state="disabled", text="Massimo 15 alimenti")

    def remove_row(self, frame):
        """Rimuove la riga corrispondente al frame passato."""
        # Trova l'indice del frame
        try:
            idx = self.row_frames.index(frame)
        except ValueError:
            return

        # Rimuovi i widget dalle liste
        del self.ingredient_entries[idx]
        del self.quantity_entries[idx]
        del self.row_frames[idx]
        del self.number_labels[idx]

        # Distruggi il frame visivo
        frame.destroy()

        # Aggiorna la numerazione delle righe successive
        for i in range(idx, len(self.number_labels)):
            self.number_labels[i].configure(text=f"{i+1}.")

        # Riabilita il pulsante "+" se eravamo al massimo e ora abbiamo meno di 15
        if len(self.ingredient_entries) < 15:
            self.add_button.configure(state="normal", text="+ Aggiungi alimento")

    def build_right_panel(self):
        """Pannello destro: tabella dei risultati."""
        ctk.CTkLabel(
            self.right_frame,
            text="Valori Nutrizionali Totali",
            font=("Arial", 16, "bold")
        ).pack(pady=10)

        table_frame = ctk.CTkFrame(self.right_frame)
        table_frame.pack(fill="both", expand=True, padx=10, pady=10)

        nutrients = [
            ("Calorie (kcal)", "calorie"),
            ("Proteine (g)",   "proteine"),
            ("Lipidi (g)",     "lipidi"),
            ("Sodio (mg)",     "sodio"),
            ("Fibre (g)",      "fibre")
        ]

        self.value_labels = {}

        for i, (name, key) in enumerate(nutrients):
            lbl_name = ctk.CTkLabel(table_frame, text=name, font=("Arial", 14))
            lbl_name.grid(row=i, column=0, padx=15, pady=8, sticky="w")

            lbl_value = ctk.CTkLabel(table_frame, text="0", font=("Arial", 14, "bold"))
            lbl_value.grid(row=i, column=1, padx=15, pady=8, sticky="e")

            self.value_labels[key] = lbl_value

        # Etichetta per messaggi di errore / conferma
        self.message_label = ctk.CTkLabel(self.right_frame, text="", text_color="red")
        self.message_label.pack(pady=(5, 10))

    def calculate(self):
        """Legge gli input, cerca gli alimenti e aggiorna i totali (arrotondati per eccesso)."""
        self.message_label.configure(text="")
        totals = {"calorie": 0.0, "proteine": 0.0, "lipidi": 0.0, "sodio": 0.0, "fibre": 0.0}
        found_any = False

        for ing_entry, qty_entry in zip(self.ingredient_entries, self.quantity_entries):
            ing = ing_entry.get().strip()
            qty_str = qty_entry.get().strip()

            if not ing and not qty_str:
                continue

            if not ing or not qty_str:
                self.message_label.configure(text="Completa entrambi i campi per ogni alimento inserito.")
                return

            try:
                qty = float(qty_str)
                if qty <= 0:
                    self.message_label.configure(text="La quantità deve essere un numero positivo.")
                    return
            except ValueError:
                self.message_label.configure(text="Inserisci una quantità numerica valida (grammi).")
                return

            food = FOOD_DB.get(ing.lower())
            if food is None:
                self.message_label.configure(text=f"Alimento '{ing}' non presente nel database.")
                return

            factor = qty / 100.0
            for key in totals:
                totals[key] += food[key] * factor

            found_any = True

        if not found_any:
            self.message_label.configure(text="Inserisci almeno un alimento per eseguire il calcolo.")
            return

        # Arrotondamento per eccesso (math.ceil) su ogni valore
        for key in totals:
            totals[key] = math.ceil(totals[key])

        for key, lbl in self.value_labels.items():
            lbl.configure(text=str(int(totals[key])))

        self.message_label.configure(text="Calcolo completato con successo.", text_color="green")


if __name__ == "__main__":
    app = NutritionCalculator()
    app.mainloop()
